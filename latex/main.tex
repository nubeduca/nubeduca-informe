% Uso: \evidencia{archivo.png}{Pie de figura}{fig:etiqueta}
\newcommand{\evidencia}[3]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{imagenes/#1}{%
      \includegraphics[width=0.95\linewidth,height=0.65\textheight,keepaspectratio]{imagenes/#1}%
    }{%
      \fbox{\parbox{0.90\linewidth}{\centering\textbf{Evidencia pendiente}\par
      Archivo esperado: \texttt{imagenes/\detokenize{#1}}}}%
    }
    \caption{#2}\label{#3}
  \end{figure}%
}
